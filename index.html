AXI4-Lite Bus Controller & Protocol Interface

A synthesizable Verilog implementation of the AMBA AXI4-Lite protocol featuring a 3-Master Priority Arbiter, 4-register Memory Block, Byte Enable Masking (WSTRB), and Error Decoding (SLVERR).

View GitHub Repository

SoC Bus Interconnect Architecture

  +---------------+
  |  Master 0     |==== Request 0 (CPU Core - High Priority) =====\
  +---------------+                                                \
  +---------------+                                                 +--------------------+      +--------------------+
  |  Master 1     |==== Request 1 (DMA Controller - Med Priority) =>| AXI4-LITE ARBITER  |=====>|  AXI Slave         |
  +---------------+                                                 |  (Bus Controller)  |      |  Memory / Regs     |
  +---------------+                                                /+--------------------+      +--------------------+
  |  Master 2     |==== Request 2 (Peripherals - Low Priority) ===/
  +---------------+

    input  wire        s_axi_arvalid,

    output reg  [31:0] s_axi_rdata,
    output reg  [1:0]  s_axi_rresp,
    output reg         s_axi_rvalid,
    input  wire        s_axi_rready
);

// axi_arbiter.v - 3-Master Priority Bus Controller
module axi_arbiter (
    input  wire       clk, rst_n,
    input  wire [2:0] req,      // M0 (CPU), M1 (DMA), M2 (Peripherals)
    output reg  [2:0] grant,    // Hot-one encoded bus grant
    input  wire       bus_busy
);
    // Priority arbitration state machine (M0 > M1 > M2)
endmodule
// axi4_lite_master.v - Initiator State Machine
module axi4_lite_master (
    input  wire clk, rst_n, start_write, start_read,
    output reg  [31:0] m_axi_awaddr, m_axi_wdata,
    input  wire m_axi_awready, m_axi_wready, m_axi_bvalid,
    output reg  done
);
    // Sequential master handshake generator logic...
endmodule
